Containerfile — ROAD RESQ AI Backend Blueprint
# ==============================================================================
# ROAD RESQ AI – Backend Application Blueprint (Containerfile / Dockerfile)
# Multi-stage container build for minimal footprint, security & production readiness.
# ==============================================================================

# ------------------------------------------------------------------------------
# STAGE 1: Dependency Builder
# ------------------------------------------------------------------------------
FROM node:20-alpine AS builder

WORKDIR /usr/src/app

# Install native compilation toolchain
RUN apk add --no-cache python3 make g++

# Copy package specifications for optimal Docker layer caching
COPY package*.json ./
RUN npm ci

# ------------------------------------------------------------------------------
# STAGE 2: Production Runtime
# ------------------------------------------------------------------------------
FROM node:20-alpine AS runner

LABEL org.opencontainers.image.title="ROAD RESQ AI - Backend Engine" \
      org.opencontainers.image.version="1.0.0" \
      org.opencontainers.image.authors="Team ROAD RESQ"

ENV NODE_ENV=production \
    PORT=5000 \
    DEFAULT_CITY_ID=chennai \
    OSRM_BASE_URL=https://router.project-osrm.org

WORKDIR /usr/src/app

RUN apk add --no-cache curl

# Copy production runtime modules
COPY package*.json ./
RUN npm ci --only=production && npm cache clean --force

# Copy backend source tree
COPY src/ ./src/

# Security hardening: Execute with non-root unprivileged user
USER node

EXPOSE 5000

# Real-time healthcheck probe
HEALTHCHECK --interval=30s --timeout=5s --start-period=10s --retries=3 \
  CMD curl -f http://localhost:5000/api/health || exit 1

CMD ["node", "src/server.js"]
